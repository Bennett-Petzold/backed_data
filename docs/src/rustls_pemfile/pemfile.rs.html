16/// The contents of a single recognised block in a PEM file.
17#[non_exhaustive]
18#[derive(Debug, PartialEq)]
19pub enum Item {
20    /// A DER-encoded x509 certificate.
21    ///
22    /// Appears as "CERTIFICATE" in PEM files.
23    X509Certificate(CertificateDer<'static>),
24
25    /// A DER-encoded Subject Public Key Info; as specified in RFC 7468.
26    ///
27    /// Appears as "PUBLIC KEY" in PEM files.
28    SubjectPublicKeyInfo(SubjectPublicKeyInfoDer<'static>),
29
30    /// A DER-encoded plaintext RSA private key; as specified in PKCS #1/RFC 3447
31    ///
32    /// Appears as "RSA PRIVATE KEY" in PEM files.
33    Pkcs1Key(PrivatePkcs1KeyDer<'static>),
34
35    /// A DER-encoded plaintext private key; as specified in PKCS #8/RFC 5958
36    ///
37    /// Appears as "PRIVATE KEY" in PEM files.
38    Pkcs8Key(PrivatePkcs8KeyDer<'static>),
39
40    /// A Sec1-encoded plaintext private key; as specified in RFC 5915
41    ///
42    /// Appears as "EC PRIVATE KEY" in PEM files.
43    Sec1Key(PrivateSec1KeyDer<'static>),
44
45    /// A Certificate Revocation List; as specified in RFC 5280
46    ///
47    /// Appears as "X509 CRL" in PEM files.
48    Crl(CertificateRevocationListDer<'static>),
49
50    /// A Certificate Signing Request; as specified in RFC 2986
51    ///
52    /// Appears as "CERTIFICATE REQUEST" in PEM files.
53    Csr(CertificateSigningRequestDer<'static>),
54}

56/// Errors that may arise when parsing the contents of a PEM file
57#[derive(Debug, PartialEq)]
58pub enum Error {
59    /// a section is missing its "END marker" line
60    MissingSectionEnd {
61        /// the expected "END marker" line that was not found
62        end_marker: Vec<u8>,
63    },
64
65    /// syntax error found in the line that starts a new section
66    IllegalSectionStart {
67        /// line that contains the syntax error
68        line: Vec<u8>,
69    },
70
71    /// base64 decode error
72    Base64Decode(String),
73}

75/// Extract and decode the next PEM section from `input`
76///
77/// - `Ok(None)` is returned if there is no PEM section to read from `input`
78/// - Syntax errors and decoding errors produce a `Err(...)`
79/// - Otherwise each decoded section is returned with a `Ok(Some((Item::..., remainder)))` where
80///   `remainder` is the part of the `input` that follows the returned section
81pub fn read_one_from_slice(mut input: &[u8]) -> Result<Option<(Item, &[u8])>, Error> {
82    let mut b64buf = Vec::with_capacity(1024);
83    let mut section = None::<(Vec<_>, Vec<_>)>;
84
85    loop {
86        let next_line = if let Some(index) = input.iter().position(|byte| *byte == b'\n') {
87            let (line, newline_plus_remainder) = input.split_at(index);
88            input = &newline_plus_remainder[1..];
89            Some(line)
90        } else {
91            None
92        };
93
94        match read_one_impl(next_line, &mut section, &mut b64buf)? {
95            ControlFlow::Continue(()) => continue,
96            ControlFlow::Break(item) => return Ok(item.map(|item| (item, input))),
97        }
98    }
99}

101/// Extract and decode the next PEM section from `rd`.
102///
103/// - Ok(None) is returned if there is no PEM section read from `rd`.
104/// - Underlying IO errors produce a `Err(...)`
105/// - Otherwise each decoded section is returned with a `Ok(Some(Item::...))`
106///
107/// You can use this function to build an iterator, for example:
108/// `for item in iter::from_fn(|| read_one(rd).transpose()) { ... }`
109#[cfg(feature = "std")]
110pub fn read_one(rd: &mut dyn io::BufRead) -> Result<Option<Item>, io::Error> {
111    let mut b64buf = Vec::with_capacity(1024);
112    let mut section = None::<(Vec<_>, Vec<_>)>;
113    let mut line = Vec::with_capacity(80);
114
115    loop {
116        line.clear();
117        let len = read_until_newline(rd, &mut line)?;
118
119        let next_line = if len == 0 {
120            None
121        } else {
122            Some(line.as_slice())
123        };
124
125        match read_one_impl(next_line, &mut section, &mut b64buf) {
126            Ok(ControlFlow::Break(opt)) => return Ok(opt),
127            Ok(ControlFlow::Continue(())) => continue,
128            Err(e) => {
129                return Err(match e {
130                    Error::MissingSectionEnd { end_marker } => io::Error::new(
131                        ErrorKind::InvalidData,
132                        format!(
133                            "section end {:?} missing",
134                            String::from_utf8_lossy(&end_marker)
135                        ),
136                    ),
137
138                    Error::IllegalSectionStart { line } => io::Error::new(
139                        ErrorKind::InvalidData,
140                        format!(
141                            "illegal section start: {:?}",
142                            String::from_utf8_lossy(&line)
143                        ),
144                    ),
145
146                    Error::Base64Decode(err) => io::Error::new(ErrorKind::InvalidData, err),
147                });
148            }
149        }
150    }
151}

153fn read_one_impl(
154    next_line: Option<&[u8]>,
155    section: &mut Option<(Vec<u8>, Vec<u8>)>,
156    b64buf: &mut Vec<u8>,
157) -> Result<ControlFlow<Option<Item>, ()>, Error> {
158    let line = if let Some(line) = next_line {
159        line
160    } else {
161        // EOF
162        return match section.take() {
163            Some((_, end_marker)) => Err(Error::MissingSectionEnd { end_marker }),
164            None => Ok(ControlFlow::Break(None)),
165        };
166    };
167
168    if line.starts_with(b"-----BEGIN ") {
169        let (mut trailer, mut pos) = (0, line.len());
170        for (i, &b) in line.iter().enumerate().rev() {
171            match b {
172                b'-' => {
173                    trailer += 1;
174                    pos = i;
175                }
176                b'\n' | b'\r' | b' ' => continue,
177                _ => break,
178            }
179        }
180
181        if trailer != 5 {
182            return Err(Error::IllegalSectionStart {
183                line: line.to_vec(),
184            });
185        }
186
187        let ty = &line[11..pos];
188        let mut end = Vec::with_capacity(10 + 4 + ty.len());
189        end.extend_from_slice(b"-----END ");
190        end.extend_from_slice(ty);
191        end.extend_from_slice(b"-----");
192        *section = Some((ty.to_owned(), end));
193        return Ok(ControlFlow::Continue(()));
194    }
195
196    if let Some((section_type, end_marker)) = section.as_ref() {
197        if line.starts_with(end_marker) {
198            let der = base64::ENGINE
199                .decode(&b64buf)
200                .map_err(|err| Error::Base64Decode(format!("{err:?}")))?;
201
202            let item = match section_type.as_slice() {
203                b"CERTIFICATE" => Some(Item::X509Certificate(der.into())),
204                b"PUBLIC KEY" => Some(Item::SubjectPublicKeyInfo(der.into())),
205                b"RSA PRIVATE KEY" => Some(Item::Pkcs1Key(der.into())),
206                b"PRIVATE KEY" => Some(Item::Pkcs8Key(der.into())),
207                b"EC PRIVATE KEY" => Some(Item::Sec1Key(der.into())),
208                b"X509 CRL" => Some(Item::Crl(der.into())),
209                b"CERTIFICATE REQUEST" => Some(Item::Csr(der.into())),
210                _ => {
211                    *section = None;
212                    b64buf.clear();
213                    None
214                }
215            };
216
217            if item.is_some() {
218                return Ok(ControlFlow::Break(item));
219            }
220        }
221    }
222
223    if section.is_some() {
224        // Extend b64buf without leading or trailing whitespace
225        b64buf.extend(trim_ascii(line));
226    }
227
228    Ok(ControlFlow::Continue(()))
229}

231// Ported from https://github.com/rust-lang/rust/blob/91cfcb021935853caa06698b759c293c09d1e96a/library/std/src/io/mod.rs#L1990 and
232// modified to look for our accepted newlines.
233#[cfg(feature = "std")]
234fn read_until_newline<R: io::BufRead + ?Sized>(r: &mut R, buf: &mut Vec<u8>) -> io::Result<usize> {
235    let mut read = 0;
236    loop {
237        let (done, used) = {
238            let available = match r.fill_buf() {
239                Ok(n) => n,
240                Err(ref e) if e.kind() == ErrorKind::Interrupted => continue,
241                Err(e) => return Err(e),
242            };
243            match available
244                .iter()
245                .copied()
246                .position(|b| b == b'\n' || b == b'\r')
247            {
248                Some(i) => {
249                    buf.extend_from_slice(&available[..=i]);
250                    (true, i + 1)
251                }
252                None => {
253                    buf.extend_from_slice(available);
254                    (false, available.len())
255                }
256            }
257        };
258        r.consume(used);
259        read += used;
260        if done || used == 0 {
261            return Ok(read);
262        }
263    }
264}

266/// Trim contiguous leading and trailing whitespace from `line`.
267///
268/// We use [u8::is_ascii_whitespace] to determine what is whitespace.
269// TODO(XXX): Replace with `[u8]::trim_ascii` once stabilized[0] and available in our MSRV.
270//   [0]: https://github.com/rust-lang/rust/issues/94035
271const fn trim_ascii(line: &[u8]) -> &[u8] {
272    let mut bytes = line;
273
274    // Note: A pattern matching based approach (instead of indexing) allows
275    // making the function const.
276    while let [first, rest @ ..] = bytes {
277        if first.is_ascii_whitespace() {
278            bytes = rest;
279        } else {
280            break;
281        }
282    }
283
284    while let [rest @ .., last] = bytes {
285        if last.is_ascii_whitespace() {
286            bytes = rest;
287        } else {
288            break;
289        }
290    }
291
292    bytes
293}

295/// Extract and return all PEM sections by reading `rd`.
296#[cfg(feature = "std")]
297pub fn read_all(rd: &mut dyn io::BufRead) -> impl Iterator<Item = Result<Item, io::Error>> + '_ {
298    iter::from_fn(move || read_one(rd).transpose())
