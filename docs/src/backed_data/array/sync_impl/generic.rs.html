34/// Read implementations
35impl<K: Container<Data = usize>, E: BackedEntryContainerNestedRead> BackedArray<K, E> {
36    #[allow(clippy::type_complexity)]
37    fn generic_get_loc(
38        &self,
39        loc: &ArrayLoc,
40        idx: usize,
41    ) -> Result<
42        BorrowNest<E::Ref<'_>, &E::Unwrapped, <E::Unwrapped as Container>::Ref<'_>>,
43        BackedArrayError<E::ReadError>,
44    > {
45        let wrapped_container = self
46            .entries
47            .c_get(loc.entry_idx)
48            .ok_or(BackedArrayError::OutsideEntryBounds(idx))?;
49
50        // Need to keep the handle to the container alive, in case it
51        // invalidates borrowed content when dropped.
52        let entry = BorrowExtender::try_new(wrapped_container, |wrapped_container| {
53            // Opening layers of abstraction, no transformations.
54            let wrapped_container: &BackedEntry<_, _, _> =
55                BackedEntryContainer::get_ref(wrapped_container.deref());
56
57            // Use the wrapped_container pointer once, to load the backing.
58            let inner_container = wrapped_container.load().map_err(BackedArrayError::Coder)?;
59
60            // `wrapped_container` is borrowed from self.entries, and held
61            // for the entire returned `'a` lifetime by BorrowExtender. The
62            // pointer that load is called against is only used once, to get an
63            // inner container value that is valid as long as
64            // `wrapped_container` is held. Transmute can then be used to
65            // extend the validity of this reference, which is valid for the
66            // lifetime of this struct.
67            //
68            // This is actually unsafe until function return. I have not found
69            // a way to make the compiler enforce the second borrow extending
70            // block.
71            Ok(unsafe { transmute::<&E::Unwrapped, &'_ E::Unwrapped>(inner_container) })
72        })?;
73
74        // Also need to keep this handle to an unwrapped container alive in
75        // the return.
76        BorrowExtender::try_new(entry, |entry| {
77            entry
78                .c_get(loc.inside_entry_idx)
79                .ok_or(BackedArrayError::OutsideEntryBounds(idx))
80        })
81    }
82
83    /// Implementor for [`Self::get`] that retains type information.
84    #[allow(clippy::type_complexity)]
85    fn internal_get(
86        &self,
87        idx: usize,
88    ) -> Result<
89        BorrowNest<E::Ref<'_>, &E::Unwrapped, <E::Unwrapped as Container>::Ref<'_>>,
90        BackedArrayError<E::ReadError>,
91    > {
92        let loc = internal_idx(
93            self.key_starts.c_ref().as_ref(),
94            self.key_ends.c_ref().as_ref(),
95            idx,
96        )
97        .ok_or(BackedArrayError::OutsideEntryBounds(idx))?;
98
99        self.generic_get_loc(&loc, idx)
100    }
101
102    /// [`Self::get`] that works on containers that don't directly dereference
103    /// to slices.
104    pub fn generic_get(
105        &self,
106        idx: usize,
107    ) -> Result<impl Deref<Target = E::InnerData> + '_, BackedArrayError<E::ReadError>> {
108        Ok(NestDeref::from(self.internal_get(idx)?))
109    }
110}

114/// Iterates over a backed array, returning each item in order.
115///
116/// To keep an accurate size count, failed reads will not be retried.
117/// This will keep each disk loaded after pulling data from it.
118/// Stepping by > 1 with `nth` implementation may skip loading a disk.
119#[derive(Debug)]
120pub struct BackedArrayGenericIter<'a, K, E> {
121    backed: &'a BackedArray<K, E>,
122    loc: ArrayLoc,
123    pos: usize,
124}
125
126impl<'a, K, E> BackedArrayGenericIter<'a, K, E> {
127    fn new(backed: &'a BackedArray<K, E>) -> Self {
128        Self {
129            backed,
130            loc: ArrayLoc {
131                entry_idx: 0,
132                inside_entry_idx: 0,
133            },
134            pos: 0,
135        }
136    }
137}
138
139impl<'a, K: Container<Data = usize>, E: BackedEntryContainerNestedRead> Iterator
140    for BackedArrayGenericIter<'a, K, E>
141{
142    type Item = Result<
143        NestDeref<E::Ref<'a>, &'a E::Unwrapped, <E::Unwrapped as Container>::Ref<'a>>,
144        E::ReadError,
145    >;
146
147    fn next(&mut self) -> Option<Self::Item> {
148        self.nth(0)
149    }
150
151    fn nth(&mut self, n: usize) -> Option<Self::Item> {
152        let cur_key_range = |this: &Self| {
153            Some(
154                *this.backed.key_ends.c_get(this.loc.entry_idx)?
155                    - *this.backed.key_starts.c_get(this.loc.entry_idx)?,
156            )
157        };
158
159        let advance = |this: &mut Self, n: usize| {
160            this.pos += n;
161            this.loc.inside_entry_idx += n;
162            while this.loc.inside_entry_idx >= cur_key_range(this)? {
163                this.loc.inside_entry_idx -= cur_key_range(this)?;
164                this.loc.entry_idx += 1;
165            }
166            Some(())
167        };
168
169        advance(self, n)?;
170
171        if self.loc.entry_idx >= self.backed.key_ends.c_len() {
172            return None;
173        }
174
175        match self.backed.generic_get_loc(&self.loc, self.pos) {
176            Ok(val) => {
177                let _ = advance(self, 1);
178                Some(Ok(NestDeref::from(val)))
179            }
180            Err(e) => match e {
181                BackedArrayError::OutsideEntryBounds(_) => None,
182                BackedArrayError::Coder(c) => {
183                    let _ = advance(self, 1);
184                    Some(Err(c))
185                }
186            },
187        }
188    }
189
190    fn size_hint(&self) -> (usize, Option<usize>) {
191        let remaining = self.backed.len() - self.pos;
192        (remaining, Some(remaining))
193    }
194}

243impl<'a, K: Container<Data = usize>, E: BackedEntryContainerNestedAll>
244    BackedArrayGenericIterMut<'a, K, E>
245{
246    fn new(backed: &'a mut BackedArray<K, E>) -> Self {
247        let key_ends = backed.key_ends.c_ref();
248        let key_ends = key_ends.as_ref();
249        let len = *key_ends.last().unwrap_or(&0);
250
251        let mut handles = Vec::with_capacity(backed.chunks_len());
252        let keys = BorrowExtender::new(
253            BorrowTuple(backed.key_starts.c_ref(), backed.key_ends.c_ref()),
254            |keys| {
255                let (key_starts, key_ends) = (&keys.0, &keys.1);
256                // The keys reference is valid as long for the life of this struct,
257                // which is valid as long as `Self` is valid.
258                let key_starts =
259                    unsafe { transmute::<&K::RefSlice<'_>, &'a K::RefSlice<'a>>(key_starts) };
260                let key_ends =
261                    unsafe { transmute::<&K::RefSlice<'_>, &'a K::RefSlice<'a>>(key_ends) };
262                key_starts.as_ref().iter().zip(key_ends.as_ref()).peekable()
263            },
264        );
265
266        let mut entries = BorrowExtender::new_mut(Box::new(backed.entries.c_mut()), |mut ent| {
267            let ent: &mut _ = unsafe { ent.as_mut() }; // Open NonNull
268
269            // The entries reference is valid as long for the life of this
270            // struct, which is valid as long as `Self` is valid.
271            let ent = unsafe { transmute::<&mut E::MutSlice<'_>, &'a mut E::MutSlice<'a>>(ent) };
272            ent.iter_mut()
273        });
274
275        if let Some(entry) = entries.by_ref().next() {
276            handles.push(Arc::new(Mutex::new(entry.get_mut().mut_handle())));
277        }
278
279        Self {
280            pos: 0,
281            len,
282            keys,
283            handles,
284            entries,
285        }
286    }
287
288    /// Flush all opened handles.
289    pub fn flush(&mut self) -> Result<&mut Self, E::WriteError> {
290        for h in self.handles.iter_mut() {
291            let mut h = h.lock().unwrap();
292            if let Ok(h) = h.deref_mut() {
293                h.flush()?;
294            }
295        }
296        Ok(self)
297    }
298}
299
300impl<K: Container, E: BackedEntryContainerNestedAll> Drop for BackedArrayGenericIterMut<'_, K, E> {
301    fn drop(&mut self) {
302        for h in self.handles.iter_mut() {
303            if let Some(h) = Arc::get_mut(h) {
304                let mut h = h.get_mut().unwrap();
305                if let Ok(h) = h.deref_mut() {
306                    h.flush()
307                        .map_err(|_| "Failed to drop handle in BackedArrayGenericIterMut!")
308                        .unwrap();
309                }
310            }
311        }
312    }
313}

315impl<'a, K: Container<Data = usize> + 'a, E: BackedEntryContainerNestedAll + 'a> Iterator
316    for BackedArrayGenericIterMut<'a, K, E>
317where
318    E::Unwrapped: 'a,
319    E::ReadError: 'a,
320    E::OnceWrapper: 'a,
321{
322    type Item = Result<
323        CountedHandle<
324            'a,
325            BackedEntry<E::OnceWrapper, E::Disk, E::Coder>,
326            <E::Unwrapped as Container>::Mut<'a>,
327        >,
328        &'a E::ReadError,
329    >;
330
331    fn next(&mut self) -> Option<Self::Item> {
332        self.nth(0)
333    }
334
335    fn nth(&mut self, n: usize) -> Option<Self::Item> {
336        self.pos += n;
337
338        // # of steps forward taken by keys
339        let mut step_count = 0;
340
341        // Remove all key positions prior to `pos`.
342        // This iterator only goes forward, so keys are not reused once
343        // passed.
344        while let Some(key) = self.keys.by_ref().peek() {
345            if *key.1 > self.pos {
346                // Update to latest entry handle
347                break;
348            }
349            let _ = self.keys.by_ref().next();
350            step_count += 1;
351        }
352
353        let inner_pos = self.pos - self.keys.peek()?.0;
354        self.pos += 1; // Position advances, even on errors
355
356        if step_count > 0 {
357            let entry = self.entries.nth(step_count - 1).unwrap().get_mut();
358            self.handles.push(Arc::new(Mutex::new(entry.mut_handle())));
359        }
360
361        let handle = self.handles.last_mut().unwrap().clone();
362        let mut handle_locked = handle.lock().unwrap();
363
364        match &mut *handle_locked {
365            Ok(h) => {
366                // The data is valid for the lifetime of the underlying
367                // handle, which is guarded by an Arc, not the iterator struct.
368                let h = unsafe {
369                    transmute::<
370                        &mut BackedEntryMut<'a, BackedEntry<E::OnceWrapper, E::Disk, E::Coder>>,
371                        &'a mut BackedEntryMut<'a, BackedEntry<E::OnceWrapper, E::Disk, E::Coder>>,
372                    >(h)
373                };
374                h.c_get_mut(inner_pos)
375                    .map(|h| Ok(CountedHandle::new(handle.clone(), h)))
376            }
377            Err(e) => {
378                // The error is valid for the lifetime of the underlying
379                // handle, which is guarded by an Arc, not the iterator struct.
380                let e = unsafe { transmute::<&E::ReadError, &'a E::ReadError>(e) };
381                Some(Err(e))
382            }
383        }
384    }
385
386    fn size_hint(&self) -> (usize, Option<usize>) {
387        let remaining = self.len - self.pos;
388        (remaining, Some(remaining))
389    }
390}

408    /// [`Self::chunk_iter`] that works on containers that don't directly
409    /// dereference to slices.
410    pub fn generic_chunk_iter(
411        &self,
412    ) -> impl Iterator<Item = Result<impl Deref<Target = &E::Unwrapped>, E::ReadError>> {
413        self.entries.ref_iter().map(|arr| {
414            BorrowExtender::try_new(arr, |arr| {
415                let loaded_entry = arr.as_ref().get_ref().load()?;
416
417                // The loaded value is valid as long as the BorrowExtender is
418                // valid, since it keeps the entry alive.
419                Ok(unsafe { transmute::<&E::Unwrapped, &'_ E::Unwrapped>(loaded_entry) })
420            })
421        })
422    }
423}

437    /// dereference to slices.
438    pub fn generic_chunk_mut_iter(
439        &mut self,
440    ) -> impl Iterator<
441        Item = Result<
442            impl Deref<Target = BackedEntryMut<'_, BackedEntry<E::OnceWrapper, E::Disk, E::Coder>>>,
443            E::ReadError,
444        >,
445    > {
446        self.entries.mut_iter().map(|arr| {
447            // Get AsRef type, keeping the entry itself alive to keep data validity
448            let arr = BorrowExtender::new_mut(arr, |mut arr| {
449                let arr: &mut _ = unsafe { arr.as_mut() }; // Open NonNull
450                open_mut!(arr)
451            });
452            BorrowExtender::try_new_mut(arr, |mut arr| unsafe { arr.as_mut() }.mut_handle())
453        })
454    }

645    fn item_mod_iter() {
646        const FIB: &[u8] = &[0, 1, 1, 2, 3, 5];
647        const INPUT_1: &[u8] = &[4, 6, 7];
648
649        let after_mod = INPUT_1
650            .iter()
651            .chain(FIB.iter().skip(INPUT_1.len()))
652            .cloned()
653            .collect::<Vec<_>>();
654        let after_second_mod = INPUT_1.iter().chain(INPUT_1).collect::<Vec<_>>();
655        let after_third_mod = INPUT_1
656            .iter()
657            .chain(INPUT_1)
658            .chain(INPUT_1)
659            .collect::<Vec<_>>();
660
661        cursor_vec!(back_vec, back_peek);
662        cursor_vec!(back_vec_2, _back_peek_3);
663
664        let mut backed = VecBackedArray::new();
665        backed
666            .append(
667                FIB,
668                unsafe { &mut *back_vec.get() },
669                BincodeCoder::default(),
670            )
671            .unwrap();
672
673        // Read checks
674        assert_eq!(
675            backed.generic_iter_mut().size_hint(),
676            (FIB.len(), Some(FIB.len()))
677        );
678        assert_eq!(backed.generic_iter_mut().count(), FIB.len());
679        assert_eq!(
680            backed
681                .generic_iter_mut()
682                .map(|x| **x.unwrap())
683                .collect::<Vec<_>>(),
684            FIB
685        );
686
687        let mut backed_iter = backed.generic_iter_mut();
688        for val in INPUT_1 {
689            let mut entry = backed_iter.next().unwrap().unwrap();
690            **entry = *val;
691        }
692
693        // Backed storage is not modified
694        #[cfg(not(miri))]
695        assert_eq!(&back_peek[back_peek.len() - FIB.len()..], FIB);
696
697        backed_iter.flush().unwrap();
698
699        // Backed storage is now modified
700        #[cfg(not(miri))]
701        assert_eq!(back_peek[back_peek.len() - after_mod.len()..], after_mod);
702
703        assert_eq!(
704            backed_iter.map(|ent| **ent.unwrap()).collect::<Vec<_>>(),
705            FIB.iter().skip(INPUT_1.len()).cloned().collect::<Vec<_>>()
706        );
707
708        assert_eq!(
709            backed
710                .generic_iter_mut()
711                .map(|ent| **ent.unwrap())
712                .collect::<Vec<_>>(),
713            after_mod
714        );
715
716        let mut backed_iter = backed.generic_iter_mut();
717        let first = **backed_iter.next().unwrap().unwrap();
718        let second = **backed_iter.next().unwrap().unwrap();
719        let third = **backed_iter.next().unwrap().unwrap();
720        assert_eq!([first, second, third], INPUT_1);
721        for val in INPUT_1 {
722            let mut entry = backed_iter.next().unwrap().unwrap();
723            **entry = *val;
724        }
725
726        #[cfg(not(miri))]
727        {
728            // USAGE FOR TESTING ONLY
729            // This breaks handle guarantees about flush on drop.
730            // Also leaks memory.
731            std::mem::forget(backed_iter);
732
733            // Backed storage is not modified
734            assert_eq!(back_peek[back_peek.len() - after_mod.len()..], after_mod);
735        }
736        #[cfg(miri)]
737        drop(backed_iter);
738
739        assert_eq!(
740            backed
741                .generic_iter_mut()
742                .map(|x| **x.unwrap())
743                .collect::<Vec<_>>(),
744            after_second_mod.iter().map(|x| **x).collect::<Vec<_>>(),
745        );
746
747        // Backed storage is updated after drop
748        #[cfg(not(miri))]
749        assert_eq!(
750            back_peek[back_peek.len() - after_mod.len()..],
751            after_second_mod.iter().map(|x| **x).collect::<Vec<_>>(),
752        );
753
754        backed
755            .append(
756                INPUT_1,
757                unsafe { &mut *back_vec_2.get() },
758                BincodeCoder::default(),
759            )
760            .unwrap();
761
762        // Correctly crosses multiple storage disks
763        assert_eq!(
764            backed
765                .generic_iter_mut()
766                .map(|x| **x.unwrap())
767                .collect::<Vec<_>>(),
768            after_third_mod.iter().map(|x| **x).collect::<Vec<_>>(),
769        );
770    }

775        let back_vector_0 = CursorVec {
776            inner: Mutex::new(&mut back_vector_0),
777        };

781        };
782
783        const INPUT_0: &[u8] = &[0, 1, 1];
784        const INPUT_1: &[u8] = &[2, 5, 7];
785
786        let mut backed = VecBackedArray::new();
787        backed
788            .append(INPUT_0, back_vector_0, BincodeCoder::default())
789            .unwrap();
790        backed
791            .append_memory(INPUT_1, back_vector_1, BincodeCoder::default())
792            .unwrap();
793
794        assert_eq!(backed.len(), 6);
795        assert_eq!(backed.loaded_len(), 3);
796        backed.shrink_to_query(&[0]);

806        });
