49pub trait RefIter<T> {
50    type IterRef<'b>: AsRef<T> + Deref<Target = T> + StableDerefFeatureOpt
51    where
52        Self: 'b;
53    fn ref_iter(&self) -> impl Iterator<Item = Self::IterRef<'_>>;
54}
55
56pub trait MutIter<T> {
57    type IterMut<'b>: AsMut<T> + DerefMut<Target = T> + StableDerefFeatureOpt
58    where
59        Self: 'b;
60    fn mut_iter(&mut self) -> impl Iterator<Item = Self::IterMut<'_>>;
61}

63/// Generic wrapper for any container type.
64///
65/// This is implemented over standard Rust containers ([`Vec`], boxed slices,
66/// etc.) in addition to more exotic types.
67/// `&[T]` is insufficiently generic for types that return a ref handle to `T`,
68/// instead of `&T` directly, so this allows for more complex container types.
69///
70/// Methods are prepended with `c_*` to avoid namespace conflicts.
71pub trait Container: RefIter<Self::Data> + MutIter<Self::Data> {
72    /// The data container entries give references to.
73    type Data;
74    type Ref<'b>: AsRef<Self::Data> + Deref<Target = Self::Data> + StableDerefFeatureOpt
75    where
76        Self: 'b;
77    type Mut<'b>: AsMut<Self::Data> + DerefMut<Target = Self::Data> + StableDerefFeatureOpt
78    where
79        Self: 'b;
80    type RefSlice<'b>: AsRef<[Self::Data]> + Deref<Target = [Self::Data]> + StableDerefFeatureOpt
81    where
82        Self: 'b;
83    type MutSlice<'b>: AsMut<[Self::Data]> + DerefMut<Target = [Self::Data]> + StableDerefFeatureOpt
84    where
85        Self: 'b;
86
87    fn c_get(&self, index: usize) -> Option<Self::Ref<'_>>;
88    fn c_get_mut(&mut self, index: usize) -> Option<Self::Mut<'_>>;
89    fn c_len(&self) -> usize;
90    fn c_ref(&self) -> Self::RefSlice<'_>;
91    fn c_mut(&mut self) -> Self::MutSlice<'_>;
92}

94/// A [`Container`] that supports resizing operations.
95pub trait ResizingContainer:
96    Container + Default + FromIterator<Self::Data> + Extend<Self::Data>
97{
98    fn c_push(&mut self, value: Self::Data);
99    fn c_remove(&mut self, index: usize);
100    fn c_append(&mut self, other: &mut Self);
101}

103/// A [`BackedEntry`] holding a valid [`Container`] type.
104///
105/// For internal implementation, reduces size of generics boilerplate.
106pub trait BackedEntryContainer {
107    type Container;
108    type Disk;
109    type Coder;
110
111    fn get_ref(&self) -> &BackedEntry<Self::Container, Self::Disk, Self::Coder>;
112    fn get_mut(&mut self) -> &mut BackedEntry<Self::Container, Self::Disk, Self::Coder>;
113    fn get(self) -> BackedEntry<Self::Container, Self::Disk, Self::Coder>;
114}
115
116impl<C, D, Enc> BackedEntryContainer for BackedEntry<C, D, Enc> {
117    type Container = C;
118    type Disk = D;
119    type Coder = Enc;
120
121    fn get(self) -> BackedEntry<Self::Container, Self::Disk, Self::Coder> {
122        self
123    }
124    fn get_ref(&self) -> &BackedEntry<Self::Container, Self::Disk, Self::Coder> {
125        self
126    }
127    fn get_mut(&mut self) -> &mut BackedEntry<Self::Container, Self::Disk, Self::Coder> {
128        self
129    }
130}

132/// A [`BackedEntryContainer`] inside a [`Container`].
133///
134/// For internal implementation, reduces size of generics boilerplate.
135pub trait BackedEntryContainerNested:
136    Container<
137    Data: BackedEntryContainer<
138        Container = Self::OnceWrapper,
139        Disk = Self::Disk,
140        Coder = Self::Coder,
141    > + From<BackedEntry<Self::OnceWrapper, Self::Disk, Self::Coder>>,
142>
143{
144    type InnerData;
145    type OnceWrapper: Once<Inner = Self::Unwrapped>;
146    type Unwrapped: Container<Data = Self::InnerData>;
147    type Disk;
148    type Coder;
149}
150
151/// Auto-implement trait to wrap intended generics.
152impl<T> BackedEntryContainerNested for T
153where
154    for<'a> T: Container<Data: BackedEntryContainer>,
155    <<T as Container>::Data as BackedEntryContainer>::Container: Once<Inner: Container>,
156    <T as Container>::Data: From<
157        BackedEntry<
158            <T::Data as BackedEntryContainer>::Container,
159            <T::Data as BackedEntryContainer>::Disk,
160            <T::Data as BackedEntryContainer>::Coder,
161        >,
162    >,
163{
164    type InnerData =
165        <<<T::Data as BackedEntryContainer>::Container as Once>::Inner as Container>::Data;
166    type OnceWrapper = <T::Data as BackedEntryContainer>::Container;
167    type Unwrapped = <<T::Data as BackedEntryContainer>::Container as Once>::Inner;
168    type Disk = <T::Data as BackedEntryContainer>::Disk;
169    type Coder = <T::Data as BackedEntryContainer>::Coder;
170}
