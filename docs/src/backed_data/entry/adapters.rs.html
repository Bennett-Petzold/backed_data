36/// Adapts a synchronous disk to be asynchronous.
37///
38/// This serializes as `inner`.
39///
40/// The function handle is used to spawn `inner`'s read/write disk on another
41/// thread. If the function handle runs on the current thread instead, this
42/// will block indefinitely when used.
43#[derive(Debug, Clone)]
44pub struct SyncAsAsync<T, RF, WF> {
45    inner: T,
46    read_handle: RF,
47    write_handle: WF,
48    read_buffer_size: usize,
49    write_buffer_min_size: usize,
50}
51
52/// Default to 8 KiB, same as [`std::io::BufReader`].
53const DEFAULT_READ_BUFFER_SIZE: usize = 8 * 1024;
54
55/// Default to 8 KiB, same as [`std::io::BufReader`].
56const DEFAULT_WRITE_BUFFER_SIZE: usize = 8 * 1024;
57
58impl<T, RF, WF> SyncAsAsync<T, RF, WF> {
59    /// Constructs a new async adapter for some [`ReadDisk`].
60    ///
61    /// # Parameters
62    /// * `inner`: [`ReadDisk`] to wrap into [`AsyncReadDisk`].
63    /// * `handle`: Handle for the background thread. MUST spawn onto another
64    ///     thread, or any read/write calls will idle block indefinitely.
65    /// * `read_buffer_size`: Size of heap allocation for `read` calls.
66    ///     Sizing this below `read`'s max size will split reads into more
67    ///     chunks, creating more heap allocations for thread transfers.
68    ///     Sizing this above `read`'s max size will result in wasted
69    ///     allocation space.
70    /// * `write_buffer_min_size`: Min heap size before `write` calls are sent
71    ///     to the background thread.
72    pub fn new(
73        inner: T,
74        read_handle: RF,
75        write_handle: WF,
76        read_buffer_size: Option<usize>,
77        write_buffer_min_size: Option<usize>,
78    ) -> Self {
79        Self {
80            inner,
81            read_handle,
82            write_handle,
83            read_buffer_size: read_buffer_size.unwrap_or(DEFAULT_READ_BUFFER_SIZE),
84            write_buffer_min_size: write_buffer_min_size.unwrap_or(DEFAULT_WRITE_BUFFER_SIZE),
85        }
86    }
87}

91    where
92        S: serde::Serializer,
93    {
94        self.inner.serialize(serializer)
95    }

98/// Runs a sync [`Read`] on a background thread for nonblocking [`AsyncRead`].
99#[derive(Debug)]
100pub struct SyncAsAsyncRead {
101    pending: bool,
102    // Provide a waker and notify with condvar to trigger a new read.
103    cmd_tx: SyncSender<ReadCmd>,
104    waker: Arc<std::sync::Mutex<Option<Waker>>>,
105    #[allow(clippy::type_complexity)]
106    read: Arc<std::sync::Mutex<Option<std::io::Result<Vec<u8>>>>>,
107    buffered: Vec<u8>,
108}
109
110#[derive(Debug)]
111enum ReadCmd {
112    Read,
113}
114
115/// [`BlockingFn`] that spawns a background reader.
116#[derive(Debug)]
117pub struct SyncAsAsyncReadBg<R> {
118    reader: R,
119    buffer_size: usize,
120    cmd_rx: Receiver<ReadCmd>,
121    waker: Arc<std::sync::Mutex<Option<Waker>>>,
122    #[allow(clippy::type_complexity)]
123    read: Arc<std::sync::Mutex<Option<std::io::Result<Vec<u8>>>>>,
124}
125
126impl<R: Read> BlockingFn for SyncAsAsyncReadBg<R> {
127    type Output = ();
128    fn call(mut self) -> Self::Output {
129        // Only valid command to be sent is a Read
130        while self.cmd_rx.recv().is_ok() {
131            let mut read_buffer = vec![0; self.buffer_size];
132
133            let res = self.reader.read(&mut read_buffer);
134
135            // Mutex guards here held until end of block
136            let mut read_send = self.read.lock().unwrap();
137
138            match res {
139                Ok(x) => {
140                    *read_send = {
141                        read_buffer.truncate(x);
142                        Some(Ok(read_buffer))
143                    }
144                }
145                Err(e) => *read_send = Some(Err(e)),
146            }
147            if let Some(w) = self.waker.lock().unwrap().take() {
148                w.wake()
149            }
150
151            drop(read_send);
152        }
153    }
154}

156impl SyncAsAsyncRead {
157    fn new<R, F>(disk: &R, handle: &F, buffer_size: usize) -> std::io::Result<Self>
158    where
159        R: ReadDisk,
160        F: Fn(SyncAsAsyncReadBg<R::ReadDisk>),
161    {
162        let (cmd_tx, cmd_rx) = sync_channel(0);
163        let waker = Arc::new(std::sync::Mutex::default());
164        let read = Arc::new(std::sync::Mutex::default());
165
166        (handle)(SyncAsAsyncReadBg {
167            reader: disk.read_disk()?,
168            cmd_rx,
169            waker: waker.clone(),
170            read: read.clone(),
171            buffer_size,
172        });
173
174        let buffered: Box<[_]> = Box::new([]);
175        let buffered = buffered.into_vec();
176
177        Ok(Self {
178            pending: false,
179            cmd_tx,
180            waker,
181            read,
182            buffered,
183        })
184    }
185}

187impl AsyncBufRead for SyncAsAsyncRead {
188    fn poll_fill_buf(
189        mut self: Pin<&mut Self>,
190        cx: &mut Context<'_>,
191    ) -> Poll<std::io::Result<&[u8]>> {
192        if self.pending {
193            // A bit of borrow checker hacking, since we can't assign to
194            // `self.buffered` while `self.read` is borrowed.
195            #[allow(unused_assignments)]
196            let mut temp_buffer = Vec::with_capacity(0);
197
198            {
199                let mut read = self.read.lock().unwrap();
200                match read.take() {
201                    Some(Ok(x)) => {
202                        temp_buffer = x;
203                    }
204                    Some(Err(e)) => {
205                        return Poll::Ready(Err(e));
206                    }
207                    None => {
208                        // Replace with the newest waker.
209                        // Since the read lock is being held, and the
210                        // processing thread writes to the read lock before
211                        // updating the waker, the `wake()` call will not race.
212                        *self.waker.lock().unwrap() = Some(cx.waker().clone());
213                        return Poll::Pending;
214                    }
215                };
216                drop(read);
217            }
218
219            self.buffered = temp_buffer;
220            self.pending = false;
221        } else if self.buffered.is_empty() {
222            self.pending = true;
223
224            *self.waker.lock().unwrap() = Some(cx.waker().clone());
225            self.cmd_tx.send(ReadCmd::Read).unwrap();
226
227            return Poll::Pending;
228        };
229
230        Poll::Ready(Ok(&self.into_ref().get_ref().buffered))
231    }
232
233    fn consume(mut self: Pin<&mut Self>, amt: usize) {
234        self.buffered.drain(..amt);
235    }
236}

238impl AsyncRead for SyncAsAsyncRead {
239    fn poll_read(
240        mut self: Pin<&mut Self>,
241        cx: &mut Context<'_>,
242        buf: &mut [u8],
243    ) -> Poll<std::io::Result<usize>> {
244        match self.as_mut().poll_fill_buf(cx) {
245            // Underlying future already handled the waker.
246            Poll::Pending => Poll::Pending,
247            Poll::Ready(x) => {
248                let x = x?;
249                let read_len = min(buf.len(), x.len());
250
251                buf[..read_len].copy_from_slice(&x[..read_len]);
252                self.consume(read_len);
253                Poll::Ready(Ok(read_len))
254            }
255        }
256    }
257}

310#[derive(Debug)]
311pub struct SyncAsAsyncWrite {
312    cmd_tx: Sender<AsyncWriteCommand>,
313    status_rx: Receiver<Result<(), WriteError>>,
314    write_buffer: Vec<u8>,
315    buffer_min_size: usize,
316    num_cmds_pending: usize,
317    retry: Option<Arc<Condvar>>,
318    retry_toggle: Arc<Mutex<Retry>>,
319    flush_waker: Arc<Mutex<Option<Waker>>>,
320}
321
322/// [`BlockingFn`] that spawns a background writer.
323#[derive(Debug)]
324pub struct SyncAsAsyncWriteBg<W> {
325    writer: W,
326    cmd_rx: Receiver<AsyncWriteCommand>,
327    status_tx: Sender<Result<(), WriteError>>,
328    retry_toggle: Arc<std::sync::Mutex<Retry>>,
329    flush_waker: Arc<Mutex<Option<Waker>>>,
330}
331
332impl<W: Write> BlockingFn for SyncAsAsyncWriteBg<W> {
333    type Output = ();
334    fn call(mut self) -> Self::Output {
335        while let Ok(cmd) = self.cmd_rx.recv() {
336            match cmd {
337                AsyncWriteCommand::Write(data) => {
338                    let mut pos = 0;
339
340                    while pos < data.len() {
341                        match self.writer.write(&data[pos..]) {
342                            Ok(advance) => {
343                                pos += advance;
344                            }
345                            // Just retry if the error is valid under async
346                            Err(error)
347                                if [ErrorKind::WouldBlock, ErrorKind::Interrupted]
348                                    .contains(&error.kind()) => {}
349                            Err(error) => {
350                                let retry_wait = Arc::new(Condvar::new());
351                                self.status_tx
352                                    .send(Err(WriteError {
353                                        error,
354                                        retry: Some(retry_wait.clone()),
355                                    }))
356                                    .unwrap();
357
358                                // Wait until sent a signal to retry from the
359                                // async controller. If the controller dropped,
360                                // also quit this receiving thread. Otherwise
361                                // only exit the wait on a non-suprious wakeup.
362                                loop {
363                                    let (mut retry_toggle, _) = retry_wait
364                                        .wait_timeout(
365                                            self.retry_toggle.lock().unwrap(),
366                                            Duration::from_secs(1),
367                                        )
368                                        .unwrap();
369
370                                    if !retry_toggle.alive {
371                                        return;
372                                    }
373
374                                    if retry_toggle.enabled {
375                                        retry_toggle.enabled = false;
376                                        break;
377                                    }
378                                }
379                            }
380                        }
381                    }
382
383                    // Send ok status for complete write
384                    let _ = self.status_tx.send(Ok(()));
385                }
386                AsyncWriteCommand::Flush => {
387                    let _ = self.status_tx.send(
388                        self.writer
389                            .flush()
390                            .map_err(|error| WriteError { error, retry: None }),
391                    );
392                    let flush_waker = self.flush_waker.lock().unwrap();
393                    if let Some(flush_waker) = &*flush_waker {
394                        flush_waker.wake_by_ref();
395                    }
396                }
397            }
398        }
399    }
400}

402impl SyncAsAsyncWrite {
403    fn new<W, F>(writer: &mut W, handle: &F, buffer_min_size: usize) -> std::io::Result<Self>
404    where
405        W: WriteDisk,
406        F: Fn(SyncAsAsyncWriteBg<W::WriteDisk>),
407    {
408        let (cmd_tx, cmd_rx) = channel();
409        let (status_tx, status_rx) = channel();
410        let retry_toggle = Arc::new(Mutex::default());
411        let flush_waker = Arc::new(Mutex::default());
412
413        (handle)(SyncAsAsyncWriteBg {
414            writer: writer.write_disk()?,
415            cmd_rx,
416            status_tx,
417            retry_toggle: retry_toggle.clone(),
418            flush_waker: flush_waker.clone(),
419        });
420
421        Ok(Self {
422            cmd_tx,
423            status_rx,
424            write_buffer: Vec::with_capacity(0),
425            buffer_min_size,
426            num_cmds_pending: 0,
427            retry: None,
428            retry_toggle,
429            flush_waker,
430        })
431    }
432}

434impl AsyncWrite for SyncAsAsyncWrite {
435    fn poll_write(
436        mut self: Pin<&mut Self>,
437        _cx: &mut Context<'_>, // This never waits.
438        buf: &[u8],
439    ) -> Poll<std::io::Result<usize>> {
440        // The underlying thread had an I/O failure, but the new write
441        // indicates the caller wants to try again. Reschedule the failed call.
442        if let Some(retry_wake) = self.retry.take() {
443            self.retry_toggle.lock().unwrap().enabled = true;
444            retry_wake.notify_one();
445        }
446
447        // Count off all the finished writes. Since the thread pauses after an
448        // error to wait for a retry, two errors will not pipeline in statuses.
449        while let Ok(status) = self.status_rx.try_recv() {
450            if let Err(e) = status {
451                self.retry = e.retry;
452                return Poll::Ready(Err(e.error));
453            } else {
454                self.num_cmds_pending -= 1;
455            }
456        }
457
458        self.write_buffer.extend_from_slice(buf);
459
460        // This check avoids constant allocation to send single bytes, assuming
461        // the writing client does not batch in large requests.
462        if self.write_buffer.len() >= self.buffer_min_size {
463            let send_buffer = std::mem::replace(&mut self.write_buffer, Vec::with_capacity(0));
464            self.cmd_tx
465                .send(AsyncWriteCommand::Write(send_buffer.into()))
466                .unwrap();
467            self.num_cmds_pending += 1;
468        }
469
470        Poll::Ready(Ok(buf.len()))
471    }
472
473    fn poll_flush(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<std::io::Result<()>> {
474        if self.flush_waker.lock().unwrap().is_none() {
475            self.cmd_tx.send(AsyncWriteCommand::Flush).unwrap();
476            self.num_cmds_pending += 1;
477        }
478
479        while let Ok(status) = self.status_rx.try_recv() {
480            if let Err(e) = status {
481                self.retry = e.retry;
482                return Poll::Ready(Err(e.error));
483            } else {
484                self.num_cmds_pending -= 1;
485            }
486        }
487
488        if self.num_cmds_pending > 0 {
489            *self.flush_waker.lock().unwrap() = Some(cx.waker().clone());
490            Poll::Pending
491        } else {
492            Poll::Ready(Ok(()))
493        }
494    }
495
496    fn poll_close(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<std::io::Result<()>> {
497        self.poll_flush(cx)
498    }
499}

501impl Drop for SyncAsAsyncWrite {
502    fn drop(&mut self) {
503        // In case the processing thread is waiting for a retry signal, set a
504        // notification that the thread died and send the final wakeup.
505        self.retry_toggle.lock().unwrap().alive = false;
506        if let Some(wake) = self.retry.take() {
507            wake.notify_all();
508        }
509    }
510}

613impl<C, D, RF, EF, DR, ER> AsyncDecoder<D::ReadDisk> for SyncCoderAsyncDisk<C, D, RF, EF, DR, ER>
614where
615    C: Decoder<Cursor<Vec<u8>>, T: Sync + Send> + Sync + Send,
616    D: AsyncReadDisk<ReadDisk: Sync + Send> + Sync + Send,
617    RF: Fn(DecodeBg<C, Cursor<Vec<u8>>>) -> DR + Sync + Send,
618    EF: Sync + Send,
619    DR: Future<Output = Result<C::T, C::Error>> + Sync + Send,
620    ER: Sync + Send,
621{
622    type Error = C::Error;
623    type T = C::T;
624
625    async fn decode(&self, mut source: D::ReadDisk) -> Result<Self::T, Self::Error> {
626        let handle = self.read_handle.as_ref().ok_or(std::io::Error::new(
627            ErrorKind::Other,
628            "Need a handle configured to run!",
629        ))?;
630        let mut buffer = Vec::new();
631        source.read_to_end(&mut buffer).await?;
632        (handle)(DecodeBg {
633            decoder: self.coder.clone(),
634            bytes: Cursor::new(buffer),
635        })
636        .await
637    }
638}

640impl<C, D, RF, EF, DR, ER> AsyncEncoder<D::WriteDisk> for SyncCoderAsyncDisk<C, D, RF, EF, DR, ER>
641where
642    C: Encoder<Vec<u8>, T: Sync + Send + Sized + Clone> + Sync + Send,
643    D: AsyncWriteDisk<WriteDisk: Sync + Send> + Sync + Send,
644    RF: Sync + Send,
645    EF: Fn(EncodeBg<C, Vec<u8>>) -> ER + Sync + Send,
646    DR: Sync + Send,
647    ER: Future<Output = Result<Vec<u8>, C::Error>> + Sync + Send,
648{
649    type Error = C::Error;
650    type T = C::T;
651
652    async fn encode(&self, data: &Self::T, mut target: D::WriteDisk) -> Result<(), Self::Error> {
653        let handle = self.encode_handle.as_ref().ok_or(std::io::Error::new(
654            ErrorKind::Other,
655            "Need a handle configured to run!",
656        ))?;
657        let encoded = (handle)(EncodeBg {
658            encoder: self.coder.clone(),
659            sink: Vec::new(),
660            data: data.clone(),
661        })
662        .await?;
663
664        target.write_all(&encoded).await?;
665        target.flush().await?;
666        target.close().await?;
667
668        Ok(())
669    }
670}
