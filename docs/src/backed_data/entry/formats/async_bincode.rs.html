7use std::marker::PhantomData;
8
9use async_bincode::futures::{AsyncBincodeReader, AsyncBincodeWriter};
10use futures::io::{AsyncRead, AsyncWrite};
11use futures::{AsyncWriteExt, SinkExt, StreamExt};
12use serde::{Deserialize, Serialize};
13
14use super::{AsyncDecoder, AsyncEncoder};
15
16#[derive(Debug, Clone, Copy, Serialize, Deserialize)]
17pub struct AsyncBincodeCoder<T: ?Sized> {
18    _phantom_data: PhantomData<T>,
19}
20
21impl<T: ?Sized> Default for AsyncBincodeCoder<T> {
22    fn default() -> Self {
23        Self {
24            _phantom_data: PhantomData,
25        }
26    }
27}
28
29impl<T, Source> AsyncDecoder<Source> for AsyncBincodeCoder<T>
30where
31    T: for<'de> Deserialize<'de> + Send + Sync,
32    Source: AsyncRead + Send + Sync + Unpin,
33{
34    type Error = bincode::Error;
35    type T = T;
36    async fn decode(&self, source: Source) -> Result<Self::T, Self::Error> {
37        AsyncBincodeReader::from(source)
38            .next()
39            .await
40            .ok_or(bincode::ErrorKind::Custom(
41                "AsyncBincodeReader stream empty".to_string(),
42            ))?
43    }
44}
45
46impl<T, Target> AsyncEncoder<Target> for AsyncBincodeCoder<T>
47where
48    T: Serialize + Send + Sync + Unpin,
49    Target: AsyncWrite + Send + Sync + Unpin,
50    for<'a> &'a mut Target: AsyncWrite,
51{
52    type Error = bincode::Error;
53    type T = T;
54    async fn encode(&self, data: &Self::T, mut target: Target) -> Result<(), Self::Error> {
55        let mut bincode_writer = AsyncBincodeWriter::from(&mut target).for_async();
56        bincode_writer.send(data).await?;
57        target.flush().await?;
58        target.close().await?;
59        Ok(())
60    }
61}
