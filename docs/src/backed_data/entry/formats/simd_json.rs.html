7use std::{
8    io::{Read, Write},
9    marker::PhantomData,

12use serde::{Deserialize, Serialize};
13
14use super::{Decoder, Encoder};

16#[cfg(feature = "serde_json")]
17use super::SerdeJsonCoder;
18
19#[derive(Debug, Default, Clone, Copy, Serialize, Deserialize)]
20pub struct SimdJsonCoder<T: ?Sized> {
21    _phantom_data: PhantomData<T>,
22}
23
24impl<T: ?Sized + for<'de> Deserialize<'de>, Source: Read> Decoder<Source> for SimdJsonCoder<T> {
25    type Error = simd_json::Error;
26    type T = T;
27    fn decode(&self, source: &mut Source) -> Result<Self::T, Self::Error> {
28        simd_json::from_reader(source)
29    }
30}
31impl<T: ?Sized + Serialize, Target: Write> Encoder<Target> for SimdJsonCoder<T>
32where
33    for<'a> &'a mut Target: Write,
34{
35    type Error = simd_json::Error;
36    type T = T;
37    fn encode(&self, data: &Self::T, mut target: Target) -> Result<(), Self::Error> {
38        simd_json::to_writer_pretty(&mut target, data)?;
39        target.flush()?;
40        Ok(())
41    }
42}
43
44#[cfg(feature = "serde_json")]
45impl<T> From<SerdeJsonCoder<T>> for SimdJsonCoder<T> {
46    fn from(_value: SerdeJsonCoder<T>) -> Self {
47        Self {
48            _phantom_data: PhantomData,
49        }
50    }
51}
