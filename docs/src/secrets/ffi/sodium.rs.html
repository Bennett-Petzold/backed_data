58/// Initialized libsodium. This function *must* be called at least once
59/// prior to using any of the other functions in this library, and
60/// callers *must* verify that it returns `true`. If it returns `false`,
61/// libsodium was unable to be properly set up and this library *must
62/// not* be used.
63///
64/// Calling it multiple times is a no-op.
65pub(crate) fn init() -> bool {
66    unsafe {
67        #[cfg(test)]
68        { if FAIL.with(|f| f.replace(false)) { return false }; let _x = 0; };
69
70        INIT.call_once(|| {
71            // NOTE: Calls to transmute fail to compile if the source
72            // and destination type have a different size. We (ab)use
73            // this fact to statically assert the size of types at
74            // compile-time.
75            //
76            // We assume that we can freely cast between rust array
77            // sizes and [`libc::size_t`]. If that's not true, DO NOT
78            // COMPILE.
79            #[allow(clippy::useless_transmute)]
80            let _ = std::mem::transmute::<usize, size_t>(0);
81
82            let mut failure = false;
83
84            // in non-development builds, ensure that core dumps are
85            // disabled
86            // This functionality exists only on UNIX systems.
87            #[cfg(unix)]
88            #[cfg(any(profile = "release", profile = "coverage"))]
89            #[cfg(not(feature = "allow-coredumps"))]
90            {
91                failure |= libc::setrlimit(libc::RLIMIT_CORE, &libc::rlimit {
92                    rlim_cur: 0,
93                    rlim_max: 0,
94                }) == -1;
95            }
96
97            // sodium_init returns 0 on success, -1 on failure, and 1 if
98            // the library is already initialized; someone else might
99            // have already initialized it before us, so we only care
100            // about failure
101            failure |= sodium_init() == -1;
102
103            INITIALIZED = !failure;
104        });
105
106        INITIALIZED
107    }
108}

110/// Allocates memory that can store `count` objects of type `T` and
111/// fills that memory with garbage bytes. Callers must ensure that they
112/// call [`sodium::free`] when this memory is no longer used.
113pub(crate) unsafe fn allocarray<T>(count: usize) -> *mut T {
114    sodium_allocarray(count, mem::size_of::<T>()).cast()
115}
116
117/// Releases memory acquired with [`sodium::allocarray`]. This function
118/// may panic if it detects that certain soundness and safety guarantees
119/// have been violated (e.g., an underflowing write).
120pub(crate) unsafe fn free<T>(ptr: *mut T) {
121    sodium_free(ptr.cast());
122}

173/// Compares `l` and `r` for equality in constant time, preventing
174/// side-channel attacks when comparing equality of secret data.
175pub(crate) fn memcmp(l: &[u8], r: &[u8]) -> bool {
176    if l.len() != r.len() {
177        return false;
178    }
179
180    unsafe {
181        sodium_memcmp(
182            l.as_ptr().cast(),
183            r.as_ptr().cast(),
184            r.len(),
185        ) == 0
186    }
187}

189/// Copies bytes from `src` to `dst` before zeroing the bytes in `src`.
190/// `dst` *must* be at least as long as `src` and *must not* overlap
191/// `src`.
192pub(crate) unsafe fn memtransfer(src: &mut [u8], dst: &mut [u8]) {
193    never!(src.len() > dst.len(),
194        "secrets: may not transfer a smaller `src` into a larger `dst`");
195
196    // Based on the requirements of `ptr::copy_nonoverlapping` and the
197    // that we're going to clobber `src`, we ensure that either
198    //
199    // * `src` is lower than `dst`, and `src` doesn't extend into `dst`, or
200    // * `dst` is lower than `src`, and `dst` doesn't extend into `src`
201    //
202    // this is proven since in safe rust, two mutable slices may not
203    // overlap with one-another
204    proven!(
205        (src.as_ptr() < dst.as_ptr() && src.as_ptr().add(src.len()) <= dst.as_ptr()) ||
206        (dst.as_ptr() < src.as_ptr() && dst.as_ptr().add(dst.len()) <= src.as_ptr()),
207        "secrets: may not transfer overlapping slices into one-another"
208    );
209
210    src.as_ptr().copy_to_nonoverlapping(dst.as_mut_ptr(), src.len());
211    memzero(src);
212}

214/// Fills `bytes` with zeroes.
215pub(crate) fn memzero(bytes: &mut [u8]) {
216    unsafe { sodium_memzero(bytes.as_mut_ptr().cast(), bytes.len()) }
217}

219/// Fills `bytes` with random bytes.
220pub(crate) fn memrandom(bytes: &mut [u8]) {
221    unsafe { randombytes_buf(bytes.as_mut_ptr().cast(), bytes.len()) }
222}
