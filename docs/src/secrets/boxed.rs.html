12/// The page protection applied to the memory underlying a [`Box`].
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14enum Prot {
15    /// Any attempt to read, write, or execute this memory will result
16    /// in a segfault.
17    NoAccess,
18
19    /// Any attempt to write to or execute the contents of this memory
20    /// will result in a segfault. Reads are permitted.
21    ReadOnly,
22
23    /// Any attempt to execute the contents of this memory will result
24    /// in a segfault. Reads and writes are permitted.
25    ReadWrite,
26}
27
28/// The type used for storing ref counts. Overflowing this type by
29/// borrowing too many times will cause a runtime panic. It seems
30/// implausible that there would be many legitimate use-cases where
31/// someone needs more than 255 simultaneous borrows of secret data.
32///
33/// TODO: Perhaps this could be moved to an associated type on a trait,
34/// such that a user who did need a larger value could provide a
35/// larger replacement.
36type RefCount = u8;
37
38/// NOTE: This implementation is not meant to be exposed directly to
39/// end-users, and user-facing wrappers must be written with care to
40/// ensure they statically enforce the required invariants. These
41/// invariants are asserted with `debug_assert` in order to catch bugs
42/// at development-time, but these assertions will be compiled out in
43/// release-mode due to the expectation that they are enforced
44/// statically.
45///
46/// TODO: document invariants
47#[derive(Eq)]
48pub(crate) struct Box<T: Bytes> {
49    /// the non-null pointer to the underlying protected memory
50    ptr: NonNull<T>,
51
52    /// the number of elements of `T` that can be stored in `ptr`
53    len: usize,
54
55    /// the pointer's current protection level
56    prot: Cell<Prot>,
57
58    /// the number of outstanding borrows; mutable borrows are tracked
59    /// here even though there is a max of one, so that asserts can
60    /// ensure invariants are obeyed
61    refs: Cell<RefCount>,
62}

64impl<T: Bytes> Box<T> {
65    /// Instantiates a new [`Box`] that can hold `len` elements of type
66    /// `T`. The callback `F` will be used for initialization and will
67    /// be called with a mutable reference to the unlocked [`Box`]. The
68    /// [`Box`] will be locked before it is returned from this function.
69    pub(crate) fn new<F>(len: usize, init: F) -> Self
70    where
71        F: FnOnce(&mut Self),
72    {
73        let mut boxed = Self::new_unlocked(len);
74
75        proven!(boxed.ptr != std::ptr::NonNull::dangling());
76        proven!(boxed.len == len);
77
78        init(&mut boxed);
79
80        boxed.lock();
81        boxed
82    }
83
84    /// Instantiates a new [`Box`] that can hold `len` elements of type
85    /// `T`. The callback `F` will be used for initialization and will
86    /// be called with a mutable reference to the unlocked [`Box`]. This
87    /// callback must return a [`Result`] indicating whether or not the
88    /// initialization succeeded (the [`Ok`] value is ignored). The
89    /// [`Box`] will be locked before it is returned from this function.
90    pub(crate) fn try_new<U, E, F>(len: usize, init: F) -> Result<Self, E>
91    where
92        F: FnOnce(&mut Self) -> Result<U, E>
93    {
94        let mut boxed = Self::new_unlocked(len);
95
96        proven!(boxed.ptr != std::ptr::NonNull::dangling());
97        proven!(boxed.len == len);
98
99        let result = init(&mut boxed);
100
101        boxed.lock();
102
103        result.map(|_| boxed)
104    }
105
106    /// Returns the number of elements in the [`Box`].
107    #[allow(clippy::missing_const_for_fn)] // not usable on min supported Rust
108    pub(crate) fn len(&self) -> usize {
109        self.len
110    }
111
112    /// Returns true if the [`Box`] is empty.
113    #[allow(clippy::missing_const_for_fn)] // not usable on min supported Rust
114    pub(crate) fn is_empty(&self) -> bool {
115        self.len == 0
116    }
117
118    /// Returns the size in bytes of the data contained in the [`Box`].
119    /// This does not include incidental metadata used in the
120    /// implementation of [`Box`] itself, only the size of the data
121    /// allocated on behalf of the user.
122    ///
123    /// It is the maximum number of bytes that can be read from the
124    /// internal pointer.
125    pub(crate) fn size(&self) -> usize {
126        self.len * T::size()
127    }
128
129    /// Allows the contents of the [`Box`] to be read from. Any call to
130    /// this function *must* be balanced with a call to
131    /// [`lock`](Box::lock). Mirroring Rust's borrowing rules, there may
132    /// be any number of outstanding immutable unlocks (technically,
133    /// limited by the max value of [`RefCount`]) *or* one mutable
134    /// unlock.
135    pub(crate) fn unlock(&self) -> &Self {
136        self.retain(Prot::ReadOnly);
137        self
138    }
139
140    /// Allows the contents of the [`Box`] to be read from and written
141    /// to. Any call to this function *must* be balanced with a call to
142    /// [`lock`](Box::lock). Mirroring Rust's borrowing rules, there may
143    /// be any number of outstanding immutable unlocks (technically,
144    /// limited by the max value of [`RefCount`]) *or* one mutable
145    /// unlock.
146    pub(crate) fn unlock_mut(&mut self) -> &mut Self {
147        self.retain(Prot::ReadWrite);
148        self
149    }
150
151    /// Disables all access to the underlying memory. Must only be
152    /// called to precisely balance prior calls to [`unlock`](Box::unlock)
153    /// and [`unlock_mut`](Box::unlock_mut).
154    ///
155    /// Calling this method in excess of the number of outstanding
156    /// unlocks will result in a runtime panic. Omitting a call to this
157    /// method and leaving an outstanding unlock will result in a
158    /// runtime panic when this object is dropped.
159    pub(crate) fn lock(&self) {
160        self.release();
161    }
162
163    /// Converts the [`Box`]'s contents into a reference. This must only
164    /// happen while it is unlocked, and the reference must go out of
165    /// scope before it is locked.
166    ///
167    /// Panics if `len == 0`, in which case it would be unsafe to
168    /// dereference the internal pointer.
169    pub(crate) fn as_ref(&self) -> &T {
170        // we use never! here to ensure that panics happen in both debug
171        // and release builds since it would be a violation of memory-
172        // safety if a zero-length dereference happens
173        never!(self.is_empty(),
174            "secrets: attempted to dereference a zero-length pointer");
175
176        proven!(self.prot.get() != Prot::NoAccess,
177            "secrets: may not call Box::as_ref while locked");
178
179        unsafe { self.ptr.as_ref() }
180    }
181
182    /// Converts the [`Box`]'s contents into a mutable reference. This
183    /// must only happen while it is mutably unlocked, and the slice
184    /// must go out of scope before it is locked.
185    ///
186    /// Panics if `len == 0`, in which case it would be unsafe to
187    /// dereference the internal pointer.
188    pub(crate) fn as_mut(&mut self) -> &mut T {
189        // we use never! here to ensure that panics happen in both debug
190        // and release builds since it would be a violation of memory-
191        // safety if a zero-length dereference happens
192        never!(self.is_empty(),
193            "secrets: attempted to dereference a zero-length pointer");
194
195        proven!(self.prot.get() == Prot::ReadWrite,
196            "secrets: may not call Box::as_mut unless mutably unlocked");
197
198        unsafe { self.ptr.as_mut() }
199    }

201    /// Converts the [`Box`]'s contents into a slice. This must only
202    /// happen while it is unlocked, and the slice must go out of scope
203    /// before it is locked.
204    pub(crate) fn as_slice(&self) -> &[T] {
205        // NOTE: after some consideration, I've decided that this method
206        // and its as_mut_slice() sister *are* safe.
207        //
208        // Using the retuned ref might cause a SIGSEGV, but this is not
209        // UB (in fact, it's explicitly defined behavior!), cannot cause
210        // a data race, cannot produce an invalid primitive, nor can it
211        // break any other guarantee of "safe Rust". Just a SIGSEGV.
212        //
213        // However, as currently used by wrappers in this crate, these
214        // methods are *never* called on unlocked data. Doing so would
215        // be indicative of a bug, so we want to detect this during
216        // development. If it happens in release mode, it's not
217        // explicitly unsafe so we don't need to enable this check.
218        proven!(self.prot.get() != Prot::NoAccess,
219            "secrets: may not call Box::as_slice while locked");
220
221        unsafe {
222            slice::from_raw_parts(
223                self.ptr.as_ptr(),
224                self.len,
225            )
226        }
227    }
228
229    /// Converts the [`Box`]'s contents into a mutable slice. This must
230    /// only happen while it is mutably unlocked, and the slice must go
231    /// out of scope before it is locked.
232    pub(crate) fn as_mut_slice(&mut self) -> &mut [T] {
233        proven!(self.prot.get() == Prot::ReadWrite,
234            "secrets: may not call Box::as_mut_slice unless mutably unlocked");
235
236        unsafe {
237            slice::from_raw_parts_mut(
238                self.ptr.as_ptr(),
239                self.len,
240            )
241        }
242    }

244    /// Instantiates a new [`Box`] that can hold `len` elements of type
245    /// `T`. This [`Box`] will be unlocked and *must* be locked before
246    /// it is dropped.
247    ///
248    /// TODO: make `len` a `NonZero` when it's stabilized and remove the
249    /// related panic.
250    fn new_unlocked(len: usize) -> Self {
251        tested!(len == 0);
252        tested!(std::mem::size_of::<T>() == 0);
253
254        assert!(sodium::init(), "secrets: failed to initialize libsodium");
255
256        // `sodium::allocarray` returns a memory location that already
257        // allows r/w access
258        let ptr = NonNull::new(unsafe { sodium::allocarray::<T>(len) })
259            .expect("secrets: failed to allocate memory");
260
261        // NOTE: We technically could save a little extra work here by
262        // initializing the struct with [`Prot::NoAccess`] and a zero
263        // refcount, and manually calling `mprotect` when finished with
264        // initialization. However, the `as_mut()` call performs sanity
265        // checks that ensure it's [`Prot::ReadWrite`] so it's easier to
266        // just send everything through the "normal" code paths.
267        Self {
268            ptr,
269            len,
270            prot: Cell::new(Prot::ReadWrite),
271            refs: Cell::new(1),
272        }
273    }
274
275    /// Performs the underlying retain half of the retain/release logic
276    /// for monitoring outstanding calls to unlock.
277    fn retain(&self, prot: Prot) {
278        let refs = self.refs.get();
279
280        tested!(refs == RefCount::min_value());
281        tested!(refs == RefCount::max_value());
282        tested!(prot == Prot::NoAccess);
283
284        if refs == 0 {
285            // when retaining, we must retain to a protection level with
286            // some access
287            proven!(prot != Prot::NoAccess,
288                "secrets: must retain readably or writably");
289
290            // allow access to the pointer and record what level of
291            // access is being permitted
292            //
293            // ordering probably doesn't matter here, but we set our
294            // internal protection flag first so we never run the risk
295            // of believing that memory is protected when it isn't
296            self.prot.set(prot);
297            mprotect(self.ptr.as_ptr(), prot);
298        } else {
299            // if we have a nonzero retain count, there is nothing to
300            // change, but we can assert some invariants:
301            //
302            //   * our current protection level *must not* be
303            //     [`Prot::NoAccess`] or we have underflowed the ref
304            //     counter
305            //   * our current protection level *must not* be
306            //     [`Prot::ReadWrite`] because that would imply non-
307            //     exclusive mutable access
308            //   * our target protection level *must* be `ReadOnly`
309            //     since otherwise would involve changing the protection
310            //     level of a currently-borrowed resource
311            proven!(Prot::NoAccess != self.prot.get(),
312                "secrets: out-of-order retain/release detected");
313            proven!(Prot::ReadWrite != self.prot.get(),
314                "secrets: cannot unlock mutably more than once");
315            proven!(Prot::ReadOnly == prot,
316                "secrets: cannot unlock mutably while unlocked immutably");
317        }
318
319        // "255 retains ought to be enough for anybody"
320        //
321        // We use `checked_add` to ensure we don't overflow our ref
322        // counter. This is ensured even in production builds because
323        // it's infeasible for consumers of this API to actually enforce
324        // this. That said, it's unlikely that anyone would need to
325        // have more than 255 outstanding retains at one time.
326        //
327        // This also protects us in the event of balanced, out-of-order
328        // retain/release code. If an out-of-order `release` causes the
329        // ref counter to wrap around below zero, the subsequent
330        // `retain` will panic here.
331        match refs.checked_add(1) {
332            Some(v)                  => self.refs.set(v),
333            None if self.is_locked() => panic!("secrets: out-of-order retain/release detected"),
334            None                     => panic!("secrets: retained too many times"),
335        };
336    }
337
338    /// Removes one outsdanding retain, and changes the memory
339    /// protection level back to [`Prot::NoAccess`] when the number of
340    /// outstanding retains reaches zero.
341    fn release(&self) {
342        // When releasing, we should always have at least one retain
343        // outstanding. This is enforced by all users through
344        // refcounting on allocation and drop.
345        proven!(self.refs.get() != 0,
346            "secrets: releases exceeded retains");
347
348        // When releasing, our protection level must allow some kind of
349        // access. If this condition isn't true, it was already
350        // [`Prot::NoAccess`] so at least the memory was protected.
351        proven!(self.prot.get() != Prot::NoAccess,
352            "secrets: releasing memory that's already locked");
353
354        // Deciding whether or not to use `checked_sub` or
355        // `wrapping_sub` here has pros and cons. The `proven!`s above
356        // help us catch this kind of accident in development, but if
357        // a released library has a bug that has imbalanced
358        // retains/releases, `wrapping_sub` will cause the refcount to
359        // underflow and wrap.
360        //
361        // `checked_sub` ensures that wrapping won't happen, but will
362        // cause consistency issues in the event of balanced but
363        // *out-of-order* calls to retain/release. In such a scenario,
364        // this will cause the retain count to be nonzero at drop time,
365        // leaving the memory unlocked for an indeterminate period of
366        // time.
367        //
368        // We choose `wrapped_sub` here because, by undeflowing, it will
369        // ensure that a subsequent `retain` will not unlock the memory
370        // and will trigger a `checked_add` runtime panic which we find
371        // preferable for safety purposes.
372        let refs = self.refs.get().wrapping_sub(1);
373
374        self.refs.set(refs);
375
376        if refs == 0 {
377            mprotect(self.ptr.as_ptr(), Prot::NoAccess);
378            self.prot.set(Prot::NoAccess);
379        }
380    }
381
382    /// Returns true if the protection level is [`NoAccess`]. Ignores
383    /// ref count.
384    fn is_locked(&self) -> bool {
385        self.prot.get() == Prot::NoAccess
386    }
387}

404impl<T: Bytes> Drop for Box<T> {
405    fn drop(&mut self) {
406        // [`Drop::drop`] is called during stack unwinding, so we may be
407        // in a panic already.
408        if !thread::panicking() {
409            // If this value is being dropped, we want to ensure that
410            // every retain has been balanced with a release. If this
411            // is not true in release, the memory will be freed
412            // momentarily so we don't need to worry about it.
413            proven!(self.refs.get() == 0,
414                "secrets: retains exceeded releases");
415
416            // Similarly, any dropped value should have previously been
417            // set to deny any access.
418            proven!(self.prot.get() == Prot::NoAccess,
419                "secrets: dropped secret was still accessible");
420        }
421
422        unsafe { sodium::free(self.ptr.as_mut()) }
423    }
424}

432impl<T: Bytes> Clone for Box<T> {
433    fn clone(&self) -> Self {
434        Self::new(self.len, |b| {
435            b.as_mut_slice().copy_from_slice(self.unlock().as_slice());
436            self.lock();
437        })
438    }

441impl<T: Bytes + ConstantEq> PartialEq for Box<T> {
442    fn eq(&self, other: &Self) -> bool {
443        if self.len != other.len {
444            return false;
445        }
446
447        let lhs = self.unlock().as_slice();
448        let rhs = other.unlock().as_slice();
449
450        let ret = lhs.constant_eq(rhs);
451
452        self.lock();
453        other.lock();
454
455        ret
456    }
457}

481/// Immediately changes the page protection level on `ptr` to `prot`.
482fn mprotect<T>(ptr: *mut T, prot: Prot) {
483    if !match prot {
484        Prot::NoAccess  => unsafe { sodium::mprotect_noaccess(ptr)  },
485        Prot::ReadOnly  => unsafe { sodium::mprotect_readonly(ptr)  },
486        Prot::ReadWrite => unsafe { sodium::mprotect_readwrite(ptr) },
487    } {
488        panic!("secrets: error setting memory protection to {:?}", prot);
489    }
490}

668        }
669
670        // this ensures that we *don't* inadvertently panic if we
671        // somehow made it through the above statement
672        for _ in 0..boxed.refs.get() {
673            boxed.lock()
674        }
675    }

706// There isn't a great way to run these tests on systems that don't have a native `fork()` call, so
707// we'll just skip them for now.
708#[cfg(all(test, target_family = "unix"))]
709mod tests_sigsegv {
710    use super::*;
711    use std::process;
712
713    fn assert_sigsegv<F>(f: F)
714    where
715        F: FnOnce(),
716    {
717        unsafe {
718            let pid      : libc::pid_t = libc::fork();
719            let mut stat : libc::c_int = 0;
720
721            match pid {
722                -1 => panic!("`fork(2)` failed"),
723                0  => { f(); process::exit(0) },
724                _  => {
725                    if libc::waitpid(pid, &mut stat, 0) == -1 {
726                        panic!("`waitpid(2)` failed");
727                    };
728
729                    // assert that the process terminated due to a signal
730                    assert!(libc::WIFSIGNALED(stat));
731
732                    // assert that we received a SIGBUS or SIGSEGV,
733                    // either of which can be sent by an attempt to
734                    // access protected memory regions
735                    assert!(
736                        libc::WTERMSIG(stat) == libc::SIGBUS ||
737                        libc::WTERMSIG(stat) == libc::SIGSEGV
738                    );
739                }
740            }
741        }
742    }
743
744    #[test]
745    fn it_kills_attempts_to_read_while_locked() {
746        assert_sigsegv(|| {
747            let val = unsafe { Box::<u32>::zero(1).ptr.as_ptr().read() };
748
749            // TODO: replace with [`test::black_box`] when stable
750            let _ = sodium::memcmp(val.as_bytes(), val.as_bytes());
751        });
752    }
753
754    #[test]
755    fn it_kills_attempts_to_write_while_locked() {
756        assert_sigsegv(|| {
757            unsafe { Box::<u64>::zero(1).ptr.as_ptr().write(1) };
758        });
759    }
760
761    #[test]
762    fn it_kills_attempts_to_read_after_explicitly_locked() {
763        assert_sigsegv(|| {
764            let boxed = Box::<u32>::random(4);
765            let val   = boxed.unlock().as_slice();
766            let _     = boxed.unlock();
767
768            boxed.lock();
769            boxed.lock();
770
771            let _ = sodium::memcmp(
772                val.as_bytes(),
773                val.as_bytes(),
774            );
775        });
776    }
