6/// Types that can be safely initialized by setting their memory to a
7/// random value.
8///
9/// The trait is marked unsafe in order to restrict implementors to
10/// types that can safely have their underlying memory randomized.
11///
12/// # Safety
13///
14/// This trait allows for overwriting a type's memory with a
15/// cryptographically random bit pattern. If it is not legal to
16/// represent your type as any potential series of bits, then your type
17/// may not implement this trait.
18pub unsafe trait Randomizable: AsContiguousBytes {
19    /// Randomizes the contents of `self`.
20    fn randomize(&mut self) {
21        sodium::memrandom(self.as_mut_bytes());
22    }
23}
24
25unsafe impl<T: AsContiguousBytes + ?Sized> Randomizable for T {}
