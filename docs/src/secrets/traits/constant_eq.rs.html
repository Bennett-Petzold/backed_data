1use crate::ffi::sodium;
2use crate::traits::*;
3
4/// A marker trait for types that can be compared for equality bitwise
5/// in constant time.
6///
7/// Note that this trait does not *force* types to be compared in
8/// constant time. When dealing with types that must be guaranteed to
9/// compare in constant time, it is highly encouraged to use wrappers
10/// that implement [`PartialEq`] by calling
11/// [`constant_eq`](ConstantEq::constant_eq).
12pub trait ConstantEq: AsContiguousBytes {
13    /// Compares `self` and `rhs`. Guaranteed to return false when the
14    /// two arguments differen in size, and guaranteed to perform the
15    /// bitwise comparison in constant O(size) time without
16    /// short-circuiting.
17    fn constant_eq(&self, rhs: &Self) -> bool {
18        sodium::memcmp(self.as_bytes(), rhs.as_bytes())
19    }
20}
21
22// Any type that can be represented as bytes can be compared in constant time.
23impl<T: AsContiguousBytes> ConstantEq for T {}
24impl<T: Bytes> ConstantEq for [T] {}
