7macro_rules! deserialize_integer_key {
8    ($method:ident => $visit:ident) => {
9        fn $method<V>(self, visitor: V) -> Result<V::Value, Self::Error>
10        where
11            V: Visitor<'de>,
12        {
13            match (self.key.parse(), self.key) {
14                (Ok(integer), _) => visitor.$visit(integer),
15                #[cfg(feature = "beef")]
16                (Err(_), s) => {
17                    if s.is_borrowed() {
18                        visitor.visit_borrowed_str(s.unwrap_borrowed())
19                    } else {
20                        visitor.visit_string(s.into_owned())
21                    }
22                }
23                #[cfg(not(feature = "beef"))]
24                (Err(_), Cow::Borrowed(s)) => visitor.visit_borrowed_str(s),
25                #[cfg(not(feature = "beef"))]
26                (Err(_), Cow::Owned(s)) => visitor.visit_string(s),
27            }
28        }
29    };
30}

119impl<'de> de::Deserializer<'de> for BorrowedCowStrDeserializer<'de> {
120    type Error = Error;
121
122    #[cfg(feature = "beef")]
123    fn deserialize_any<V>(self, visitor: V) -> Result<V::Value, Self::Error>
124    where
125        V: de::Visitor<'de>,
126    {
127        if self.value.is_borrowed() {
128            visitor.visit_borrowed_str(self.value.unwrap_borrowed())
129        } else {
130            visitor.visit_string(self.value.into_owned())
131        }
132    }
133    #[cfg(not(feature = "beef"))]
134    fn deserialize_any<V>(self, visitor: V) -> Result<V::Value, Self::Error>
135    where
136        V: de::Visitor<'de>,
137    {
138        match self.value {
139            Cow::Borrowed(string) => visitor.visit_borrowed_str(string),
140            Cow::Owned(string) => visitor.visit_string(string),
141        }
142    }
143
144    fn deserialize_enum<V>(
145        self,
146        _name: &str,
147        _variants: &'static [&'static str],
148        visitor: V,
149    ) -> Result<V::Value, Self::Error>
150    where
151        V: de::Visitor<'de>,
152    {
153        visitor.visit_enum(self)
154    }
155
156    forward_to_deserialize_any! {
157        bool i8 i16 i32 i64 i128 u8 u16 u32 u64 u128 f32 f64 char str string
158        bytes byte_buf option unit unit_struct newtype_struct seq tuple
159        tuple_struct map struct identifier ignored_any
160    }
161}
