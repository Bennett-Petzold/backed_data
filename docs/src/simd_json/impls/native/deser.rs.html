7#[allow(clippy::cast_possible_truncation)]
8pub(crate) unsafe fn parse_str<'invoke, 'de>(
9    input: SillyWrapper<'de>,
10    data: &'invoke [u8],
11    _buffer: &'invoke mut [u8],
12    idx: usize,
13) -> Result<&'de str> {
14    use ErrorType::{InvalidEscape, InvalidUnicodeCodepoint};
15
16    let input = input.input;
17    // skip leading `"`
18    let src: &[u8] = data.get_kinda_unchecked(idx + 1..);
19    let input = input.add(idx + 1);
20
21    let mut src_i = 0;
22    let mut b = *src.get_kinda_unchecked(src_i);
23
24    // quickly skip all the "good stuff"
25    while b != b'"' && b != b'\\' {
26        src_i += 1;
27        b = *src.get_kinda_unchecked(src_i);
28    }
29    if b == b'"' {
30        let v = std::str::from_utf8_unchecked(std::slice::from_raw_parts(input, src_i));
31        return Ok(v);
32    }
33
34    // we hit an escape sequence now it gets tricky
35    // our destination index is idx +1 to skip the '"' + our current position
36    let mut dst_i = src_i;
37    while b != b'"' {
38        if b == b'\\' {
39            // don't advance i yet
40            let escape_char = *src.get_kinda_unchecked(src_i + 1);
41            if escape_char == b'u' {
42                // got to reduce by 1 since we have to include the '\\' for get_unicode_codepoint
43                let (cp, src_offset) = get_unicode_codepoint(src.get_kinda_unchecked(src_i..))
44                    .map_err(|_| {
45                        Deserializer::error_c(idx + 1 + src_i, 'u', InvalidUnicodeCodepoint)
46                    })?;
47
48                // from  codepoint_to_utf8 since we write directly to input
49                if cp <= 0x7F {
50                    input.add(dst_i).write(cp as u8);
51                    dst_i += 1;
52                } else if cp <= 0x7FF {
53                    input.add(dst_i).write(((cp >> 6) + 192) as u8);
54                    dst_i += 1;
55                    input.add(dst_i).write(((cp & 63) + 128) as u8);
56                    dst_i += 1;
57                } else if cp <= 0xFFFF {
58                    input.add(dst_i).write(((cp >> 12) + 224) as u8);
59                    dst_i += 1;
60                    input.add(dst_i).write((((cp >> 6) & 63) + 128) as u8);
61                    dst_i += 1;
62                    input.add(dst_i).write(((cp & 63) + 128) as u8);
63                    dst_i += 1;
64                } else if cp <= 0x0010_FFFF {
65                    input.add(dst_i).write(((cp >> 18) + 240) as u8);
66                    dst_i += 1;
67                    input.add(dst_i).write((((cp >> 12) & 63) + 128) as u8);
68                    dst_i += 1;
69                    input.add(dst_i).write((((cp >> 6) & 63) + 128) as u8);
70                    dst_i += 1;
71                    input.add(dst_i).write(((cp & 63) + 128) as u8);
72                    dst_i += 1;
73                } else {
74                    return Err(Deserializer::error_c(
75                        idx + 1 + src_i,
76                        'u',
77                        InvalidUnicodeCodepoint,
78                    ));
79                }
80                // We have to subtract one since we're already moving to the next character at the end of the loop
81                src_i += src_offset - 1;
82            } else {
83                let escape_result: u8 = *ESCAPE_MAP.get_kinda_unchecked(escape_char as usize);
84                if escape_result == 0 {
85                    return Err(Deserializer::error_c(
86                        idx + 1 + src_i,
87                        escape_char as char,
88                        InvalidEscape,
89                    ));
90                }
91                input.add(dst_i).write(escape_result);
92                dst_i += 1;
93                // move i for reading the escape char
94                src_i += 1;
95            }
96        } else {
97            input.add(dst_i).write(b);
98            dst_i += 1;
99        }
100        src_i += 1;
101        b = *src.get_kinda_unchecked(src_i);
102    }
103    Ok(std::str::from_utf8_unchecked(std::slice::from_raw_parts(
104        input, dst_i,
105    )))
106}
