18#[allow(clippy::if_not_else, clippy::cast_possible_wrap)]
19#[cfg_attr(not(feature = "no-inline"), inline)]
20pub(crate) unsafe fn parse_str<'invoke, 'de>(
21    input: SillyWrapper<'de>,
22    data: &'invoke [u8],
23    buffer: &'invoke mut [u8],
24    mut idx: usize,
25) -> Result<&'de str> {
26    use ErrorType::{InvalidEscape, InvalidUnicodeCodepoint};
27    let input = input.input;
28    // Add 1 to skip the initial "
29    idx += 1;
30
31    // we include the terminal '"' so we know where to end
32    // This is safe since we check sub's length in the range access above and only
33    // create sub sliced form sub to `sub.len()`.
34
35    let src: &[u8] = data.get_kinda_unchecked(idx..);
36    let mut src_i: usize = 0;
37    let mut len = src_i;
38    loop {
39        // _mm_loadu_si128 does not require alignment
40        #[allow(clippy::cast_ptr_alignment)]
41        let v: __m128i =
42            unsafe { _mm_loadu_si128(src.as_ptr().add(src_i).cast::<arch::__m128i>()) };
43
44        // store to dest unconditionally - we can overwrite the bits we don't like
45        // later
46        let bs_bits: u32 = unsafe {
47            static_cast_u32!(_mm_movemask_epi8(_mm_cmpeq_epi8(
48                v,
49                _mm_set1_epi8(b'\\' as i8)
50            )))
51        };
52        let quote_mask = _mm_cmpeq_epi8(v, _mm_set1_epi8(b'"' as i8));
53        let quote_bits = static_cast_u32!(_mm_movemask_epi8(quote_mask));
54        if (bs_bits.wrapping_sub(1) & quote_bits) != 0 {
55            // we encountered quotes first. Move dst to point to quotes and exit
56            // find out where the quote is...
57            let quote_dist: u32 = quote_bits.trailing_zeros();
58
59            ///////////////////////
60            // Above, check for overflow in case someone has a crazy string (>=4GB?)
61            // But only add the overflow check when the document itself exceeds 4GB
62            // Currently unneeded because we refuse to parse docs larger or equal to 4GB.
63            ////////////////////////
64
65            // we advance the point, accounting for the fact that we have a NULl termination
66
67            len += quote_dist as usize;
68            let v = std::str::from_utf8_unchecked(std::slice::from_raw_parts(input.add(idx), len));
69            return Ok(v);
70
71            // we compare the pointers since we care if they are 'at the same spot'
72            // not if they are the same value
73        }
74        if (quote_bits.wrapping_sub(1) & bs_bits) == 0 {
75            // they are the same. Since they can't co-occur, it means we encountered
76            // neither.
77            src_i += 16;
78            len += 16;
79        } else {
80            // Move to the 'bad' character
81            let bs_dist: u32 = bs_bits.trailing_zeros();
82            len += bs_dist as usize;
83            src_i += bs_dist as usize;
84            break;
85        }
86    }
87
88    let mut dst_i: usize = 0;
89
90    // To be more conform with upstream
91    loop {
92        // _mm_loadu_si128 does not require alignment
93        #[allow(clippy::cast_ptr_alignment)]
94        let v: __m128i = _mm_loadu_si128(src.as_ptr().add(src_i).cast::<arch::__m128i>());
95
96        // _mm_storeu_si128 does not require alignment
97        #[allow(clippy::cast_ptr_alignment)]
98        _mm_storeu_si128(buffer.as_mut_ptr().add(dst_i).cast::<arch::__m128i>(), v);
99
100        // store to dest unconditionally - we can overwrite the bits we don't like
101        // later
102        let bs_bits: u32 = static_cast_u32!(_mm_movemask_epi8(_mm_cmpeq_epi8(
103            v,
104            _mm_set1_epi8(b'\\' as i8)
105        )));
106        let quote_mask = _mm_cmpeq_epi8(v, _mm_set1_epi8(b'"' as i8));
107        let quote_bits = static_cast_u32!(_mm_movemask_epi8(quote_mask));
108        if (bs_bits.wrapping_sub(1) & quote_bits) != 0 {
109            // we encountered quotes first. Move dst to point to quotes and exit
110            // find out where the quote is...
111            let quote_dist: u32 = quote_bits.trailing_zeros();
112
113            ///////////////////////
114            // Above, check for overflow in case someone has a crazy string (>=4GB?)
115            // But only add the overflow check when the document itself exceeds 4GB
116            // Currently unneeded because we refuse to parse docs larger or equal to 4GB.
117            ////////////////////////
118
119            // we advance the point, accounting for the fact that we have a NULl termination
120
121            dst_i += quote_dist as usize;
122            input
123                .add(idx + len)
124                .copy_from_nonoverlapping(buffer.as_ptr(), dst_i);
125            let v = std::str::from_utf8_unchecked(std::slice::from_raw_parts(
126                input.add(idx),
127                len + dst_i,
128            ));
129            return Ok(v);
130
131            // we compare the pointers since we care if they are 'at the same spot'
132            // not if they are the same value
133        }
134        if (quote_bits.wrapping_sub(1) & bs_bits) != 0 {
135            // find out where the backspace is
136            let bs_dist: u32 = bs_bits.trailing_zeros();
137            let escape_char: u8 = *src.get_kinda_unchecked(src_i + bs_dist as usize + 1);
138            // we encountered backslash first. Handle backslash
139            if escape_char == b'u' {
140                // move src/dst up to the start; they will be further adjusted
141                // within the unicode codepoint handling code.
142                src_i += bs_dist as usize;
143                dst_i += bs_dist as usize;
144                let (o, s) = if let Ok(r) = handle_unicode_codepoint(
145                    src.get_kinda_unchecked(src_i..),
146                    buffer.get_kinda_unchecked_mut(dst_i..),
147                ) {
148                    r
149                } else {
150                    return Err(Deserializer::error_c(src_i, 'u', InvalidUnicodeCodepoint));
151                };
152                if o == 0 {
153                    return Err(Deserializer::error_c(src_i, 'u', InvalidUnicodeCodepoint));
154                };
155                // We moved o steps forward at the destination and 6 on the source
156                src_i += s;
157                dst_i += o;
158            } else {
159                // simple 1:1 conversion. Will eat bs_dist+2 characters in input and
160                // write bs_dist+1 characters to output
161                // note this may reach beyond the part of the buffer we've actually
162                // seen. I think this is ok
163                let escape_result: u8 = *ESCAPE_MAP.get_kinda_unchecked(escape_char as usize);
164                if escape_result == 0 {
165                    return Err(Deserializer::error_c(
166                        src_i,
167                        escape_char as char,
168                        InvalidEscape,
169                    ));
170                }
171                *buffer.get_kinda_unchecked_mut(dst_i + bs_dist as usize) = escape_result;
172                src_i += bs_dist as usize + 2;
173                dst_i += bs_dist as usize + 1;
174            }
175        } else {
176            // they are the same. Since they can't co-occur, it means we encountered
177            // neither.
178            src_i += 16;
179            dst_i += 16;
180        }
181    }
