1//! A version of the reaper that waits for a signal to check for process progress.

3use async_lock::{Mutex as AsyncMutex, MutexGuard as AsyncMutexGuard};
4use event_listener::Event;
5use futures_lite::future;

7use std::io;
8use std::mem;
9use std::sync::Mutex;

11pub(crate) type Lock = AsyncMutexGuard<'static, ()>;
12
13/// The zombie process reaper.
14pub(crate) struct Reaper {
15    /// An event delivered every time the SIGCHLD signal occurs.
16    sigchld: Event,
17
18    /// The list of zombie processes.
19    zombies: Mutex<Vec<std::process::Child>>,
20
21    /// The pipe that delivers signal notifications.
22    pipe: Pipe,
23
24    /// Locking this mutex indicates that we are polling the SIGCHLD event.
25    driver_guard: AsyncMutex<()>,
26}
27
28impl Reaper {
29    /// Create a new reaper.
30    pub(crate) fn new() -> Self {
31        Reaper {
32            sigchld: Event::new(),
33            zombies: Mutex::new(Vec::new()),
34            pipe: Pipe::new().expect("cannot create SIGCHLD pipe"),
35            driver_guard: AsyncMutex::new(()),
36        }
37    }
38
39    /// Lock the driver thread.
40    pub(crate) async fn lock(&self) -> AsyncMutexGuard<'_, ()> {
41        self.driver_guard.lock().await
42    }
43
44    /// Reap zombie processes forever.
45    pub(crate) async fn reap(&'static self, _driver_guard: async_lock::MutexGuard<'_, ()>) -> ! {
46        loop {
47            // Wait for the next SIGCHLD signal.
48            self.pipe.wait().await;
49
50            // Notify all listeners waiting on the SIGCHLD event.
51            self.sigchld.notify(usize::MAX);
52
53            // Reap zombie processes, but make sure we don't hold onto the lock for too long!
54            let mut zombies = mem::take(&mut *self.zombies.lock().unwrap());
55            let mut i = 0;
56            'reap_zombies: loop {
57                for _ in 0..50 {
58                    if i >= zombies.len() {
59                        break 'reap_zombies;
60                    }
61
62                    if let Ok(None) = zombies[i].try_wait() {
63                        i += 1;
64                    } else {
65                        zombies.swap_remove(i);
66                    }
67                }
68
69                // Be a good citizen; yield if there are a lot of processes.
70                //
71                // After we yield, check if there are more zombie processes.
72                future::yield_now().await;
73                zombies.append(&mut self.zombies.lock().unwrap());
74            }
75
76            // Put zombie processes back.
77            self.zombies.lock().unwrap().append(&mut zombies);
78        }
79    }
80
81    /// Register a process with this reaper.
82    pub(crate) fn register(&'static self, child: std::process::Child) -> io::Result<ChildGuard> {
83        self.pipe.register(&child)?;
84        Ok(ChildGuard { inner: Some(child) })
85    }
86
87    /// Wait for an event to occur for a child process.
88    pub(crate) async fn status(
89        &'static self,
90        child: &Mutex<crate::ChildGuard>,
91    ) -> io::Result<std::process::ExitStatus> {
92        loop {
93            // Wait on the child process.
94            if let Some(status) = child.lock().unwrap().get_mut().try_wait()? {
95                return Ok(status);
96            }
97
98            // Start listening.
99            event_listener::listener!(self.sigchld => listener);
100
101            // Try again.
102            if let Some(status) = child.lock().unwrap().get_mut().try_wait()? {
103                return Ok(status);
104            }
105
106            // Wait on the listener.
107            listener.await;
108        }
109    }
110
111    /// Do we have any registered zombie processes?
112    pub(crate) fn has_zombies(&'static self) -> bool {
113        !self
114            .zombies
115            .lock()
116            .unwrap_or_else(|x| x.into_inner())
117            .is_empty()
118    }
119}
120
121/// The wrapper around the child.
122pub(crate) struct ChildGuard {
123    inner: Option<std::process::Child>,
124}
125
126impl ChildGuard {
127    /// Get a mutable reference to the inner child.
128    pub(crate) fn get_mut(&mut self) -> &mut std::process::Child {
129        self.inner.as_mut().unwrap()
130    }
131
132    /// Begin the reaping process for this child.
133    pub(crate) fn reap(&mut self, reaper: &'static Reaper) {
134        if let Ok(None) = self.get_mut().try_wait() {
135            reaper
136                .zombies
137                .lock()
138                .unwrap()
139                .push(self.inner.take().unwrap());
140        }
141    }
142}

155            },
156        };
157
158        /// Waits for the next SIGCHLD signal.
159        struct Pipe {
160            /// The sender channel for the SIGCHLD signal.
161            sender: Sender<()>,
162
163            /// The receiver channel for the SIGCHLD signal.
164            receiver: Receiver<()>,
165        }
166
167        impl Pipe {
168            /// Creates a new pipe.
169            fn new() -> io::Result<Pipe> {
170                let (sender, receiver) = bounded(1);
171                Ok(Pipe {
172                    sender,
173                    receiver
174                })
175            }
176
177            /// Waits for the next SIGCHLD signal.
178            async fn wait(&self) {
179                self.receiver.recv().await.ok();
180            }
181
182            /// Register a process object into this pipe.
183            fn register(&self, child: &std::process::Child) -> io::Result<()> {
184                // Called when a child exits.
185                #[allow(clippy::infallible_destructuring_match)]
186                unsafe extern "system" fn callback(_: *mut c_void, _: BOOLEAN) {
187                    let reaper = match &crate::Reaper::get().sys {
188                        super::Reaper::Signal(reaper) => reaper,
189                    };
190
191                    reaper.pipe.sender.try_send(()).ok();
192                }
193
194                // Register this child process to invoke `callback` on exit.
195                let mut wait_object = ptr::null_mut();
196                let ret = unsafe {
197                    RegisterWaitForSingleObject(
198                        &mut wait_object,
199                        child.as_raw_handle() as HANDLE,
200                        Some(callback),
201                        std::ptr::null_mut(),
202                        INFINITE,
203                        WT_EXECUTEINWAITTHREAD | WT_EXECUTEONLYONCE,
204                    )
205                };
206
207                if ret == 0 {
208                    Err(io::Error::last_os_error())
209                } else {
210                    Ok(())
211                }
212            }
213        }
214    } else if #[cfg(unix)] {

216        use futures_lite::prelude::*;

219        struct Pipe {
